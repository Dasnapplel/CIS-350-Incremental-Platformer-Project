 24    def draw_upgrades(self, window):
 25        # Draws the background image
 26        window.blit(self.background, (0, 0))
 27        # Labels the upgrade shop at the top of the window
 28        menu_text = pygame.font.Font(None, 36).render("Upgrade Shop", True, "#b68f40")
 29        menu_rect = menu_text.get_rect(center=(settings.WINDOW_WIDTH//2, 50))
 30        window.blit(menu_text, menu_rect)
 31        # Displays the player's points in the top right corner
 32        points = pygame.font.Font(None, 28).render(f"Points: {self.points}", True, (255,255,255))
 33        window.blit(points, points.get_rect(topright=(window.get_width() - 10, 10)))
 34
 35        # draws every upgrade in a grid
 36        for upgrade in self.upgrades:
 37            upgrade.update(window)
 38            name = pygame.font.Font(None, 36).render(upgrade.name, True, (255, 255, 255))
 39            window.blit(name, name.get_rect(center=(upgrade.x_pos, upgrade.y_pos - 5)))
 40            upgrade.draw_tallies((upgrade.x_pos, upgrade.y_pos + 20), window)
 41
 42            # Draws cost of the upgrade
 43            if upgrade.current_level < upgrade.max_level:
 44                cost_text = pygame.font.Font(None, 22).render(f"{upgrade.cost}", True, (255, 215, 0))
 45            else:
 46                cost_text = pygame.font.Font(None, 18).render("MAX", True, (255, 0, 0))
 47            window.blit(cost_text, cost_text.get_rect(right = upgrade.rect.right -14, centery = upgrade.y_pos + 30))
 48            # Draws the back button
 49            self.back_button.update(window)

 57    def purchase(self, position):
 58        # checks for clicks on the upgrade buttons and purchases if possible
 59        for upgrade in self.upgrades:
 60            if upgrade.checkForInput(position):
 61                if upgrade.current_level < upgrade.max_level and self.points >= upgrade.cost:
 62                    self.points -= upgrade.cost
 63                    upgrade.level_up()
 64                    return upgrade

 70class Upgrade(button.Button):
 71    def __init__(self, pos, name, cost, max_level):
 72        self.image = pygame.image.load(get_asset_path("upgrade.png"))
 73        self.scale(1)
 74        super().__init__(self.image, pos, "", pygame.font.Font(None, 36), (200,200,200), (150,150,150))
 75        self.name = name
 76        self.cost = cost
 77        self.max_level = max_level
 78        self.current_level = 0
 79
 80    def scale(self, scale):
 81        # Resizes the upgrade button image based on the given scale factor
 82        self.image = pygame.transform.scale(self.image, (200*scale, 100*scale))
 83
 84    def level_up(self):
 85        if self.current_level < self.max_level:
 86            self.current_level += 1
 87
 88    def draw_tallies(self, position, window):
 89        tally_height = 12
 90        tally_width = 6
 91        for i in range(self.max_level):
 92            tally = pygame.Rect(0, 0, tally_width, tally_height)
 93            # Calculates the center x of each tally based on the total number of tallies
 94            tally.centerx = (i+0.5)*(tally_width + tally_width//1.2) - (self.max_level*(tally_width + tally_width//1.2))//2 + position[0] 
 95            tally.y = position[1]
 96            if i < self.current_level:
 97                pygame.draw.rect(window, (255, 215, 0), tally)
 98            else:
 99                pygame.draw.rect(window, (150,150,150), tally)

101def upgrade_shop_main(UpgradeWindow, player):
102    upgrade_shop = UpgradeWindow
103    window = pygame.display.set_mode((settings.WINDOW_WIDTH, settings.WINDOW_HEIGHT))
104    pygame.display.flip()
105    running = True
106    while running:
107        for event in pygame.event.get():
108            if event.type == pygame.QUIT:
109                running = False
110            elif event.type == pygame.MOUSEBUTTONDOWN:
111                upgrade = upgrade_shop.purchase(pygame.mouse.get_pos())
112                if upgrade:
113                    if upgrade.name == "Double Jump":
114                        player.max_jumps = 2
115                    elif upgrade.name == "Dash":
116                        player.can_dash = True
117                    elif upgrade.name == "Speed Boost":
118                        player.speed += 25
119                        upgrade.cost += 10
120                    elif upgrade.name == "Wall Jump":
121                        player.can_wall_jump = True
122                if upgrade_shop.back_button.checkForInput(pygame.mouse.get_pos()):
123                    running = False
124
125        window.fill(settings.BACKGROUND)
126        upgrade_shop.draw_upgrades(window)
127        pygame.display.flip()
128
129if __name__ == "__main__":
130    player = player.Player(pygame.Rect(75,500,30,30))
131    upgrade_shop_main(UpgradeWindow(100000), player)
